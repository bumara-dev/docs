---
title: "ZRA Customer Sync Implementation Plan"
description: "Register each customer with ZRA on create and update, using best-effort semantics so customer mutations never fail because of a ZRA outage."
---

**Goal:** Automatically register each customer with ZRA via `POST /branches/saveBrancheCustomers` when the customer is created or updated, with best-effort failure semantics so customer mutations never fail because of a ZRA outage.

**Architecture:** Four new columns on `customers` track sync state. A new fire-and-forget service `syncCustomerToZra` runs via `c.executionCtx.waitUntil(...)` after `createCustomerHandler` and `updateCustomerHandler` return their responses. A manual `POST /invoicing/customers/{id}/sync-zra` endpoint awaits the same service for retries from the UI.

**Tech Stack:** Drizzle ORM (Postgres), TypeScript, Cloudflare Workers (Hono + zod-openapi), Vitest with `vi.hoisted` mocks.

**Spec:** [docs/superpowers/specs/2026-06-08-zra-customer-sync-design.md](/superpowers/specs/2026-06-08-zra-customer-sync-design)

**Note on commits:** The user is handling commits manually at the end. Each task notes what should be staged, but does not run `git commit`.

 * Updates ZRA sync status fields on a customer record.
 *
 * Bypasses the regular updateCustomer service which has TPIN-uniqueness

 *
 * @param {DatabaseClient} db - Drizzle database client
 * @param {string} customerId - Customer ID
 * @param {string} organizationId - Organization ID (used as a guard to ensure ownership)
 * @param {object} patch - Fields to set
 */
export async function updateCustomerZraSyncStatus(
  db: DatabaseClient,
  customerId: string,
  organizationId: string,
  patch: {
    zraSyncStatus?: 'synced' | 'failed' | 'skipped_no_device';

    .update(customers)
    .set({
      ...patch,
      updatedAt: new Date(),

        eq(customers.organizationId, organizationId)
      )
    );
}

```typescript
import {
  findCustomerById,
  findVsdcDevice,
  updateCustomerZraSyncStatus,
} from "@repo/database/repositories";
import type { ServiceContext, ServiceDependencies } from "../../../core/context";
import { requireOrganizationContext } from "../../../core/context";
import { ServiceError } from "../../../core/errors";
import { ZraApiClient, type ZraBranchCustomerPayload } from "./zra-api-client";

const DEFAULT_SANDBOX_URL = "https://api-sandbox.zra.org.zm/vsdc-api/v1";

export interface CustomerSyncResult {
  status: "synced" | "failed" | "skipped_no_device";
  error?: string;
}

export async function syncCustomerToZra(
  ctx: ServiceContext,
  deps: ServiceDependencies,
  customerId: string
): Promise<CustomerSyncResult> {
  const orgId = requireOrganizationContext(ctx);

  const customer = await findCustomerById(deps.db, customerId, orgId);
  if (!customer) {
    throw new ServiceError("NOT_FOUND", "Customer not found");
  }

  const device = await findVsdcDevice(deps.db, orgId);
  if (!device || !device.isInitialized) {
    await updateCustomerZraSyncStatus(deps.db, customerId, orgId, {
      zraSyncStatus: "skipped_no_device",
      zraSyncAttemptedAt: new Date(),
    });
    return { status: "skipped_no_device" };
  }

  const actorId = ctx.userId ?? "SYSTEM";
  const payload: ZraBranchCustomerPayload = {
    tpin: device.tpin ?? "",
    bhfId: device.branchId ?? "000",
    custNo: customer.customerNumber,
    custTpin: customer.tpin ?? "",
    custNm: customer.companyName ?? customer.name,
    adrs: customer.billingAddress ?? null,
    telNo: customer.phone ?? null,
    email: customer.email ?? null,
    faxNo: null,
    useYn: customer.isActive ? "Y" : "N",
    remark: null,
    regrNm: actorId,
    regrId: actorId,
    modrNm: actorId,
    modrId: actorId,
  };

  const client = new ZraApiClient({
    baseUrl: device.baseUrl ?? device.vsdcEndpointUrl ?? DEFAULT_SANDBOX_URL,
    apiKey: device.apiKey ?? undefined,
    tpin: device.tpin ?? undefined,
    branchId: device.branchId ?? undefined,
    deviceSerialNumber: device.deviceSerialNumber,
  });

  const apiResult = await client.saveBranchCustomer(payload);

  if (apiResult.success) {
    await updateCustomerZraSyncStatus(deps.db, customerId, orgId, {
      zraSyncStatus: "synced",
      zraSyncedAt: new Date(),
      zraSyncAttemptedAt: new Date(),
      zraSyncError: null,
    });
    return { status: "synced" };
  }

  const errorMessage = apiResult.error ?? apiResult.message;
  await updateCustomerZraSyncStatus(deps.db, customerId, orgId, {
    zraSyncStatus: "failed",
    zraSyncAttemptedAt: new Date(),
    zraSyncError: errorMessage,
  });
  return { status: "failed", error: errorMessage };
}
```

  CreateCustomerRoute

    const ctx = buildServiceContext(c);
    const deps = buildServiceDependencies(c);
    const body = await c.req.json<CreateCustomerInput>();
    const customer = await createCustomer(ctx, deps, body);
